===== STRUCTURE_DESCRIPTION =====
name: mockingjay_dead
doc: >
  Mockingjay with bypass_reject and dead_table. To address should_bypass and kept_dead,
  we added a 2-bit dead_table tracking lines that are never reused (evicted from sampler
  or age > 95). If dead_table >= 2 and bypass_reject allows it, the line is bypassed
  or inserted with ETR=11. To fit dead_table within the 48KB budget, we reduced the
  width of set_clock and timestamp from 8 to 7 bits, which is safe since entries
  are freed at age 95. This preserves the 4-bit etr_clock decay rate while preventing
  wrong bypasses via the bypass_reject table.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}
      words: {bypass_is_safe: {reject_table: bypass_reject, sig: pc_sig}}
    - table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}
      words: {bypass_is_safe: {reject_table: bypass_reject, sig: pc_sig}}
    - table: {table: dead_table, index: pc_sig, cmp: gte, threshold: 2}
      words: {bypass_is_safe: {reject_table: bypass_reject, sig: pc_sig}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_reject
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO
  - name: dead_table
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: dead_table, index: signature, op: decrement}
      - when: {table: {table: rdp, index: signature, cmp: gt, threshold: 73}}
        train: {table: bypass_reject, index: signature, op: increment}
      - when: {table: {table: rdp, index: signature, cmp: lte, threshold: 73}}
        train: {table: bypass_reject, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_reject, index: signature, op: decrement}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: dead_table, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: bypass_reject, index: signature, op: decrement}
      - train: {table: dead_table, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: dead_table, index: pc_sig, cmp: gte, threshold: 2}, words: {bypass_is_safe: {reject_table: bypass_reject, sig: pc_sig}}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}, words: {bypass_is_safe: {reject_table: bypass_reject, sig: pc_sig}}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: bypass_is_safe
  place: condition
  params: {reject_table: table, sig: signature}
  means: >
    Reads the entry of the reject_table at the index sig. Returns true if the
    entry's value is less than 2, and false otherwise. Nothing is written.
