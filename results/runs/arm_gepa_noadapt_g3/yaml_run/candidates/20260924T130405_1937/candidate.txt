===== STRUCTURE_DESCRIPTION =====
name: mockingjay_88
doc: >
  Capacity-aligned Mockingjay: raises the absolute bypass and maximum prediction
  thresholds from 80 to 88 (11 * 8), perfectly aligning with maximum ETR (11) and
  the 12-way LLC capacity (12 * 8 = 96 accesses per set). In a 12-way cache, lines
  with reuse distance between 81 and 88 easily fit, but were previously rejected by
  threshold 80, causing thousands of wrong bypasses in milc, leslie3d, and bwaves.
  Retains the 4096-entry RDP to minimize aliasing, cold-safe initialization (init_rd = 48),
  and streamlines on_match by removing dead prefetch handling.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO
  - name: init_rd
    scope: global
    kind: counter
    width: 7
    init: 48

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 88}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 4
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 6}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - when: {table: {table: rdp, index: signature, cmp: absent}}
        train: {table: rdp, index: signature, op: toward, sample: init_rd}
        stop: true
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  update_clocks:
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}

  apply_prediction:
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 88}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - sample: sampled_cache
  - run: update_clocks
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 0}}
    stop: true
  - run: apply_prediction

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - sample: sampled_cache
  - run: update_clocks
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 5}}
    stop: true
  - run: apply_prediction
