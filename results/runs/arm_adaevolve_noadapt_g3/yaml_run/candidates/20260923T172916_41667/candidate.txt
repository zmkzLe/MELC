===== STRUCTURE_DESCRIPTION =====
name: mockingjay_decoupled_rdp
doc: >
  STRATEGY: Zero-Based Hardware Budgeting and Pruning.
  Pruned `bypass_sel` and set-dueling bypass logic (reclaimed 10 bits), and
  shrunk `etr` from 5 bits to 4 bits (-8..7) (reclaimed 6 KB).
  With the reclaimed 6 KB budget, introduced a fundamentally new mechanism:
  Decoupled Reuse Distance Prediction for Demand vs. Prefetch accesses.
  Added `rdp_prefetch` table (3.5 KB) and expanded `rdp_demand` to 4096 entries
  (3.5 KB). The sampler now tracks `is_pref` to train the correct table upon
  eviction or match. This prevents prefetch accesses from polluting demand RDP.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 4                  # holds -8..7
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7                  # TIMESTAMP_BITS; wraps at 128
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {type_in: [PREFETCH], table: {table: rdp_prefetch, index: pc_sig, cmp: gt, threshold: 73}}
    - {type_in: [PREFETCH], safe_bypass: {table: rdp_prefetch, index: pc_sig, div: 12, field: etr, min_rd: 36}}
    - {type_not_in: [PREFETCH], table: {table: rdp_demand, index: pc_sig, cmp: gt, threshold: 73}}
    - {type_not_in: [PREFETCH], safe_bypass: {table: rdp_demand, index: pc_sig, div: 12, field: etr, min_rd: 48}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12

tables:
  - name: rdp_demand
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_prefetch
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 16
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
      - {name: is_pref, kind: flag, width: 1, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {counter: {counter: is_pref, cmp: eq, threshold: 1}}
        train: {table: rdp_prefetch, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp_demand, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        words: {train_expired: {table_pref: rdp_prefetch, table_dem: rdp_demand}}
        free: true
    on_replace:
      - words: {train_expired: {table_pref: rdp_prefetch, table_dem: rdp_demand}}
    install:
      signature: {signature: pc_sig}
      timestamp: {read: set_clock}
      is_pref: {words: {get_is_pref: {}}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: gt, threshold: -8}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp_prefetch, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {type_in: [PREFETCH], table: {table: rdp_prefetch, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 7}}
      stop: true
    - when: {type_in: [PREFETCH]}
      set: {etr: {table: rdp_prefetch, index: pc_sig, div: 12}}
      stop: true
    - when: {table: {table: rdp_demand, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp_demand, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 7}}
      stop: true
    - set: {etr: {table: rdp_demand, index: pc_sig, div: 12}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: gt, threshold: -8}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp_prefetch, index: pc_sig, cmp: absent}}
      set: {etr: {const: 7}}
      stop: true
    - when: {type_in: [PREFETCH], table: {table: rdp_prefetch, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 7}}
      stop: true
    - when: {type_in: [PREFETCH]}
      set: {etr: {table: rdp_prefetch, index: pc_sig, div: 12}}
      stop: true
    - when: {table: {table: rdp_demand, index: pc_sig, cmp: absent}}
      set: {etr: {const: 5}}
      stop: true
    - when: {table: {table: rdp_demand, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 7}}
      stop: true
    - set: {etr: {table: rdp_demand, index: pc_sig, div: 12}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -8}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====


- name: safe_bypass
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}}
  means: >
    True when the entry of the table at the index is present, its value is greater
    than or equal to the min_rd parameter, its value divided by the div parameter
    (integer division, rounding down) is strictly greater than the current value
    of the per_line state field at the way chosen by the victim search, and that
    field's value is greater than or equal to 0. False otherwise. Only valid inside
    victim.bypass_when.

- name: train_expired
  place: step
  params: {table_pref: table, table_dem: table}
  means: >
    Reads the is_pref field of the current sampler entry. If it is 1, increments
    the entry of the table_pref table at the index given by the entry's signature
    field. If it is 0, increments the entry of the table_dem table at the index
    given by the entry's signature field. In both cases, the increment is clamped
    at the table's maximum value.

- name: get_is_pref
  place: value
  params: {}
  means: >
    Returns 1 if the access type of the current access is PREFETCH, and 0
    otherwise. The returned value is an unsigned integer of width 1.
