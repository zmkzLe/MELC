===== STRUCTURE_DESCRIPTION =====
# Fix dead-line retention and wrong bypasses: introduce guarded_bypass which
# only permits bypass when the chosen victim is NOT overdue (ETR >= 0). When a
# victim is already overdue (ETR < 0), bypassing is disallowed so the dead line
# is evicted and the incoming line is admitted at ETR 11, preventing thousands
# of wrong bypasses on streaming/high-RD workloads while keeping active lines safe.
name: mockingjay_guarded_t16
doc: >
  Mockingjay with guarded bypass and cold-line pollution prevention.
  Replaces unconditional capacity bypass and safe_bypass with a unified
  guarded_bypass condition. Bypass is strictly prohibited if the victim chosen
  by the search is overdue (ETR < 0), ensuring dead lines are never protected at
  the expense of incoming lines. When the victim is alive (ETR >= 0), incoming
  lines with RD > 73 or lines with RD >= 40 exceeding the victim's predicted
  remaining reuse are safely bypassed. Unknown fills receive ETR 8 to prevent
  cold-line pollution, while unknown hits receive ETR 0. Retains 16-bit sampler
  tags with 3-bit etr_clock and 7-bit set_clock within the 48 KB budget.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7                  # TIMESTAMP_BITS; wraps at 128, safe since max RD is 95
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {guarded_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 40}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 16
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 8}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====


- name: guarded_bypass
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}}
  means: >
    True when the entry of the table at the index is present, the per_line state
    field at the way chosen by the victim search is greater than or equal to 0,
    and either the table entry is strictly greater than 73, or the table entry is
    greater than or equal to the min_rd parameter and its value divided by the div
    parameter (integer division, rounding down) is strictly greater than that
    state field's value. False otherwise. Only valid inside victim.bypass_when.
