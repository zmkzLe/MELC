===== STRUCTURE_DESCRIPTION =====
name: mockingjay_combined_v3
doc: >
  Combines the clamped_dec aging fix (preventing lines with ETR=11 from getting
  stuck and causing kept_dead errors) with access-type aware dynamic bypass.
  Prefetches use more aggressive bypass thresholds (min_rd=32, force_rd=70)
  while demand accesses use conservative thresholds (min_rd=40, force_rd=74).
  Set dueling arbitrates bypass globally to prevent wrong_bypass on hostile traces.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7
    init: ZERO
  - name: bypass_sel
    scope: global
    kind: saturating
    width: 10
    init: MID

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {type_in: [PREFETCH], dynamic_bypass_eval: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 32, force_rd: 70, selector: bypass_sel}}
    - {type_not_in: [PREFETCH], dynamic_bypass_eval: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 40, force_rd: 74, selector: bypass_sel}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 16
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        set: {etr: {clamped_dec: {field: etr, min: -11}}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        set: {etr: {clamped_dec: {field: etr, min: -11}}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 8}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_not_in: [WRITE]}
    words: {dynamic_bypass_update: {selector: bypass_sel}}
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====

- name: dynamic_bypass_eval
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}, force_rd: {int: [0, 127]}, selector: state}
  means: >
    Evaluates the core bypass condition: true if the table entry at the index is
    present, AND either (its value >= force_rd) OR (its value >= min_rd AND
    its value / div > the current value of the per_line state field at the
    victim way AND that field's value >= 0).
    The final result depends on ChampSim's get_set_sample_category(set) c:
    If c == 1, returns false (Leader A: never bypass).
    If c == 2, returns the core condition (Leader B: bypass enabled).
    Otherwise, returns the core condition ONLY IF the global saturating selector
    state field is >= its midpoint (e.g., 512 for a 10-bit counter).
    Only valid inside victim.bypass_when.
- name: dynamic_bypass_update
  place: step
  params: {selector: state}
  means: >
    Reads ChampSim's get_set_sample_category(set) c.
    If c == 1, increments the global saturating selector state field (clamped at max).
    If c == 2, decrements the selector field (clamped at 0).
    Otherwise, does nothing.
- name: clamped_dec
  place: value
  params: {field: state, min: {int: [-16, 0]}}
  means: >
    Reads the per_line state field at the visited way. If its signed value
    is greater than min, returns its value minus 1. Otherwise returns min.
