===== STRUCTURE_DESCRIPTION =====
# Fix aggressive bypass and cold-line pollution: introduce a confidence table (conf)
# to track the accuracy of the Reuse Distance Predictor (RDP). The sampler trains
# conf by comparing the actual age with the predicted RD. The bypass_when conditions
# now require a minimum confidence (min_conf: 2) before allowing a bypass, which
# directly addresses the massive 'wrong_bypass' penalty. Tag bits in the sampler
# are reduced to 14 to afford the 2-bit conf table within the 48 KB limit.
name: mockingjay_conf_bypass
doc: >
  Mockingjay with confidence-gated bypass to prevent wrong_bypass mistakes.
  Introduces a 2-bit confidence table (conf) indexed by pc_sig. The sampler
  trains conf: increments when the predicted RD matches the actual sampled age
  within a tolerance, and decrements otherwise. The bypass_when conditions now
  require a minimum confidence (min_conf: 2) before allowing a bypass. Tag bits
  in the sampler are reduced to 14 to afford the conf table within the 48 KB limit.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7                  # TIMESTAMP_BITS; wraps at 128
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {confident_high_rd: {rdp: rdp, conf: conf, index: pc_sig, rd_threshold: 73, min_conf: 2}}
    - {safe_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 36, conf_table: conf, min_conf: 2}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 14
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - words: {train_conf: {table: conf, rdp_table: rdp, index: signature, age: age, tolerance: 16}}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        words: {train_conf: {table: conf, rdp_table: rdp, index: signature, age: age, tolerance: 16}}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - words: {train_conf: {table: conf, rdp_table: rdp, index: signature, age: age, tolerance: 16}}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 8}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====

- name: train_conf
  place: step
  params: {table: table, rdp_table: table, index: signature, age: state, tolerance: {int: [0, 127]}}
  means: >
    Reads the value of the rdp_table at the index. If it is present, calculates
    the absolute difference between its value and the age parameter. If the
    rdp_table value is 95 and the age parameter is greater than or equal to 95,
    the difference is considered 0. If the difference is less than or equal to
    the tolerance parameter, increments the entry of the table at the index
    (clamped at its maximum). Otherwise, decrements the entry of the table at
    the index (clamped at 0).

- name: confident_high_rd
  place: condition
  params: {rdp: table, conf: table, index: signature, rd_threshold: {int: [0, 127]}, min_conf: {int: [0, 7]}}
  means: >
    True when the entry of the rdp table at the index is present and its value
    is strictly greater than the rd_threshold parameter, and the entry of the
    conf table at the index is greater than or equal to the min_conf parameter.
    False otherwise.

- name: safe_bypass
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}, conf_table: table, min_conf: {int: [0, 7]}}
  means: >
    True when the entry of the table at the index is present, its value is greater
    than or equal to the min_rd parameter, its value divided by the div parameter
    (integer division, rounding down) is strictly greater than the current value
    of the per_line state field at the way chosen by the victim search, that
    field's value is greater than or equal to 0, and the entry of the conf_table
    at the index is greater than or equal to the min_conf parameter. False otherwise.
    Only valid inside victim.bypass_when.
