===== STRUCTURE_DESCRIPTION =====
name: mockingjay_duel_bypass_v4
doc: >
  Dynamic Bypass Arbitration via Set Dueling with optimized ETR magnitudes.
  Bypassing is globally arbitrated by a saturating counter tracking demand misses.
  Leader A (category 1) never bypasses; Leader B (category 2) always bypasses
  when RDP is high. Followers follow the selector.
  ETR magnitudes are optimized: dead lines get ETR 15 (evicted first), writebacks
  get ETR 13 (evicted second), and overdue lines stop at -12 (evicted third).
  This fixes magnitude inversions and perfectly aligns with the bypass condition,
  which never protects dead or overdue lines.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -16..15
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7                  # TIMESTAMP_BITS; wraps at 128
    init: ZERO
  - name: bypass_sel
    scope: global
    kind: saturating
    width: 10
    init: MID

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {duel_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 36, force_rd: 74, selector: bypass_sel}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 16
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 12, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 15}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 12, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 15}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 8}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 15}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_not_in: [WRITE, PREFETCH]}
    words: {update_bypass_duel: {selector: bypass_sel}}
  - when: {type_in: [WRITE]}
    set: {etr: {const: 13}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====


- name: duel_bypass
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}, force_rd: {int: [0, 127]}, selector: state}
  means: >
    Evaluates the core bypass condition: true if the table entry at the index is
    present, AND either (its value >= force_rd) OR (its value >= min_rd AND
    its value / div > the current value of the per_line state field at the
    victim way AND that field's value >= 0).
    The final result depends on ChampSim's get_set_sample_category(set) c:
    If c == 1, returns false (Leader A: never bypass).
    If c == 2, returns the core condition (Leader B: bypass enabled).
    Otherwise, returns the core condition ONLY IF the global saturating selector
    state field is >= its midpoint (e.g., 512 for a 10-bit counter).
    Only valid inside victim.bypass_when.

- name: update_bypass_duel
  place: step
  params: {selector: state}
  means: >
    Reads ChampSim's get_set_sample_category(set) c.
    If c == 1, increments the global saturating selector state field (clamped at max).
    If c == 2, decrements the selector field (clamped at 0).
    Otherwise, does nothing.
