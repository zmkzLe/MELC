===== STRUCTURE_DESCRIPTION =====
# Fix aging freeze on ETR 11 lines and tune safe bypass threshold: introduce
# can_age to allow lines at ETR 11 to properly age down (fixing a bug where
# abs(etr) < 11 prevented +11 lines from ever aging), set safe_bypass min_rd to
# 48 to eliminate thousands of wrong bypasses on moderate reuse distances, and
# retain 13-bit sampler tags with 3-bit etr_clock under the 48 KB budget.
name: mockingjay_safe_t13_fix
doc: >
  Enhanced Mockingjay with bug-free aging and conservative safe bypass.
  Replaces the buggy abs(etr) < 11 aging filter with can_age, allowing lines
  at ETR +11 to age down rather than remaining frozen at +11 indefinitely.
  Uses safe_bypass with min_rd 48 so lines with reuse distances up to 48
  are never bypassed, protecting resident cache lines while avoiding wrong
  bypasses on active working sets. Fills with absent predictions are placed
  at ETR 8 to prevent dead cold-line retention, and unknown hits get ETR 0.
  Sampler accuracy is maximized with 13-bit tags enabled by a 3-bit etr_clock.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7                   # GRANULARITY: ages set every 8 accesses (wraps 7->0)
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {safe_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 48}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 13
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {can_age: {field: etr}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {can_age: {field: etr}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 8}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====


- name: safe_bypass
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}}
  means: >
    True when the entry of the table at the index is present, its value is greater
    than or equal to the min_rd parameter, its value divided by the div parameter
    (integer division, rounding down) is strictly greater than the current value
    of the per_line state field at the way chosen by the victim search, and that
    field's value is greater than or equal to 0. False otherwise. Only valid inside
    victim.bypass_when.

- name: can_age
  place: condition
  params: {field: state}
  means: >
    True when the signed per_line state field is strictly greater than -11.
    False when it is less than or equal to -11.
