===== STRUCTURE_DESCRIPTION =====
name: mockingjay_duel_bypass_v3
doc: >
  Dynamic Bypass Arbitration via Set Dueling for moderate RD.
  Lines with RD >= 74 are always bypassed. For lines with moderate RD (>= 36),
  bypassing is dynamically enabled/disabled via set dueling. Leader sets A (category 1)
  never moderate-bypass, leader sets B (category 2) always moderate-bypass.
  Follower sets moderate-bypass if the global saturating counter favors it.
  This avoids wrong_bypass on hostile traces while keeping benefits.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7
    init: ZERO
  - name: bypass_sel
    scope: global
    kind: saturating
    width: 10
    init: MID

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {duel_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 36, force_rd: 74, selector: bypass_sel}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 16
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 8}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_not_in: [WRITE]}
    words: {update_bypass_duel: {selector: bypass_sel}}
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====

- name: duel_bypass
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}, force_rd: {int: [0, 127]}, selector: state}
  means: >
    Evaluates the bypass condition. First, if the table entry at the index is
    present and its value is greater than or equal to force_rd, returns true.
    Otherwise, evaluates the moderate bypass condition: true if the entry is
    present, its value is greater than or equal to min_rd, its value divided
    by div (integer division) is strictly greater than the current value of the
    per_line state field at the victim way, and that field's value is greater
    than or equal to 0. If the moderate bypass condition is false, returns false.
    If it is true, the final result depends on ChampSim's get_set_sample_category(set) c:
    if c == 1, returns false; if c == 2, returns true; otherwise, returns true
    only if the global saturating selector state field is greater than or equal
    to its midpoint (e.g., 512 for a 10-bit counter). Only valid inside victim.bypass_when.
- name: update_bypass_duel
  place: step
  params: {selector: state}
  means: >
    Reads ChampSim's get_set_sample_category(set) c.
    If c == 1, increments the global saturating selector state field (clamped at its maximum).
    If c == 2, decrements the selector field (clamped at 0).
    Otherwise, does nothing.
