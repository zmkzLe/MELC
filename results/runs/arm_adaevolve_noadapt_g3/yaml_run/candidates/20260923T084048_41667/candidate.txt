===== STRUCTURE_DESCRIPTION =====
# Fix aggressive bypass and cold-line pollution: introduce safe_bypass_v2 to only
# bypass if incoming RD >= 40, is worse than the victim, and the victim is NOT
# overdue (stops protecting dead lines). Unknown fills get ETR 4 to prevent
# cold-line pollution, while unknown prefetch fills get ETR 11. Unknown hits get ETR 0.
# Optimized etr_clock to 3 bits to afford 13-bit tags in the sampler.
name: mockingjay_safe_v2
doc: >
  Mockingjay with safe bypass and cold-line pollution prevention.
  Uses a safe_bypass_v2 condition to only bypass if incoming RD >= 40, is worse
  than the victim, and the victim is not overdue (ETR >= 0).
  Unknown hits get ETR 0. Unknown fills get ETR 4 to prevent cold-line pollution
  without keeping them dead for too long. Unknown prefetch fills get ETR 11.
  Optimizes etr_clock to 3 bits (wrapping automatically) to afford 13-bit tags.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {safe_bypass_v2: {table: rdp, index: pc_sig, div: 8, field: etr, min_rd: 40}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 13
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  hit_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  fill_access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 4}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: hit_access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: fill_access
===== NEW_WORDS =====

- name: safe_bypass_v2
  place: condition
  params: {table: table, index: signature, div: {int: [1, 16]}, field: state, min_rd: {int: [0, 127]}}
  means: >
    True when the entry of the table at the index is present, its value is greater
    than or equal to the min_rd parameter, its value divided by the div parameter
    (integer division, rounding down) is strictly greater than the current value
    of the per_line state field at the way chosen by the victim search, and that
    field's value is greater than or equal to 0. False otherwise. Only valid inside
    victim.bypass_when.
