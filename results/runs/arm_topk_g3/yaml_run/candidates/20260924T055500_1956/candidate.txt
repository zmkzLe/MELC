===== STRUCTURE_DESCRIPTION =====
name: mockingjay_conf
doc: >
  Mockingjay with a dynamic bypass confidence threshold to balance wrong_bypass
  and should_bypass. A 3-bit bypass_conf table tracks the probability that a line
  is dead (evicted or age > 95 before reuse). Bypasses predicted by RDP only occur
  if confidence >= 4. Highly confident dead lines (confidence == 7) are bypassed
  regardless of RDP. etr_clock width is reduced to 3 bits to save space, wrapping
  naturally every 8 accesses.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 7
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}, table_check: {table: bypass_conf, index: pc_sig, cmp: gte, threshold: 4}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, table_check: {table: bypass_conf, index: pc_sig, cmp: gte, threshold: 4}}
    - {table_check: {table: bypass_conf, index: pc_sig, cmp: eq, threshold: 7}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 3
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: bypass_conf, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        train: {table: bypass_conf, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: bypass_conf, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: table_check
  place: condition
  params: {table: table, index: signature, cmp: {symbol: Comparison}, threshold: {int: [0, 127]}}
  means: >
    True when the entry of the table at the index is present and its value satisfies the comparison with the threshold; an absent entry counts as 0. Nothing is written.
