===== STRUCTURE_DESCRIPTION =====
name: mockingjay_conf
doc: >
  Added a bypass confidence table to penalize the bypass confidence of a signature
  when a wrong_bypass occurs, as suggested by the feedback. Bypasses only occur
  if confidence is >= 2. Confidence is updated in the sampler: penalized on wrong
  predictions and rewarded on correct ones.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}, check_conf: {table: bypass_conf, index: pc_sig, limit: 2}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, check_conf: {table: bypass_conf, index: pc_sig, limit: 2}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    max: 3
    init: 3

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train_bypass_conf_on_reuse: {conf_table: bypass_conf, rdp_table: rdp, sig: signature, threshold: 73}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train_bypass_conf_on_dead: {conf_table: bypass_conf, rdp_table: rdp, sig: signature, threshold: 73}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access

===== NEW_WORDS =====

- name: check_conf
  place: condition
  params: {index: signature, limit: {int: [0, 3]}, table: table}
  means: >
    True when the entry of the table at the index is present and its value is
    greater than or equal to the limit.

- name: train_bypass_conf_on_reuse
  place: step
  params: {conf_table: table, rdp_table: table, sig: signature, threshold: {int: [0, 127]}}
  means: >
    Reads the entry of rdp_table at the index sig. If the entry is present and
    its value is strictly greater than threshold, decrements the entry of
    conf_table at the index sig, clamping at 0. Otherwise, increments the entry
    of conf_table at the index sig, clamping at its maximum.

- name: train_bypass_conf_on_dead
  place: step
  params: {conf_table: table, rdp_table: table, sig: signature, threshold: {int: [0, 127]}}
  means: >
    Reads the entry of rdp_table at the index sig. If the entry is present and
    its value is strictly greater than threshold, increments the entry of
    conf_table at the index sig, clamping at its maximum. Otherwise, decrements
    the entry of conf_table at the index sig, clamping at 0.
