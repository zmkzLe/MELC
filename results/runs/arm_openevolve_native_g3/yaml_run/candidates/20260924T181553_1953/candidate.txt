name: mockingjay_ema_conf
doc: >
  Improved Mockingjay with EMA for RDP training and confidence-gated bypassing.
  Fixes massive wrong_bypass by requiring confidence even for table_exceeds_victim,
  and adds a new word punish_wrong_bypass to instantly reset confidence when a
  predicted-dead line is reused.

state:

    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, confident_bypass: {conf: conf, conf_thresh: 2, index: pc_sig, rdp: rdp, rdp_thresh: 0}}
    - {confident_bypass: {conf: conf, conf_thresh: 2, index: pc_sig, rdp: rdp, rdp_thresh: 73}}

  - name: conf

    age: {clock: set_clock, stamp: timestamp}

    on_match:

      - when: {type_in: [PREFETCH]}

    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:

===== NEW_WORDS =====

- name: punish_wrong_bypass
  place: step
  params:
    index: signature
    rdp: table
    rdp_thresh:
      int: [0, 127]
    conf: table
  means: >
    Reads the entry of the rdp table at the index. If it is present and greater than rdp_thresh,
    sets the entry of the conf table at the index to 0 to heavily penalize the wrong bypass prediction.
    Otherwise, decrements the conf table entry by 1 (clamping at 0).

- name: confident_bypass
  place: condition
  params:
    conf: table
    conf_thresh:
      int: [0, 3]
    index: signature
    rdp: table
    rdp_thresh:
      int: [0, 127]
  means: 'Reads the entry of the rdp table at the index. If it is absent, returns false. Otherwise, reads
    the entry of the conf table at the index (treating an absent entry as 0). Returns true if the rdp
    entry is greater than rdp_thresh and the conf entry is greater than or equal to conf_thresh.

    '

- name: train_ema
  place: step
  params:
    index: signature
    sample: state
    scale:
      int: [1, 4]
    shift:
      int: [0, 7]
    table: table
  means: 'Calculates a target value as the sample multiplied by the scale, clamped to the table''s maximum.
    Reads the entry of the table at the index. If it is absent, sets it to the target value. If present,
    computes the signed difference (target - entry), divides it by 2 to the power of shift (integer division
    rounding towards zero), and adds the result to the entry. The final value is clamped to the table''s
    minimum (0) and maximum.

    '

- name: train_penalty
  place: step
  params:
    index: signature
    penalty:
      int: [1, 127]
    table: table
  means: Reads the entry of the table at the index. If it is absent, sets it to the table's maximum. If
    present, adds the penalty to the entry, clamping the result to the table's maximum.
