===== STRUCTURE_DESCRIPTION =====
name: mockingjay_opt
doc: >
  Addresses wrong_bypass by introducing penalize_wrong_bypass, a step that heavily
  penalizes the RDP table when a line is prematurely requested (sample <= threshold
  but RDP > threshold). Addresses should_bypass and kept_dead by using a custom
  train_penalty step to aggressively increase the RDP of lines that see no reuse,
  ensuring they reach the bypass threshold faster and are evicted sooner.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - words: {penalize_wrong_bypass: {table: rdp, index: signature, sample: age, threshold: 73, penalty: 32}}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        words: {train_penalty: {table: rdp, index: signature, penalty: 8}}
        free: true
    on_replace:
      - words: {train_penalty: {table: rdp, index: signature, penalty: 8}}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: penalize_wrong_bypass
  place: step
  params: {table: table, index: signature, sample: state, threshold: {int: [0, 127]}, penalty: {int: [1, 127]}}
  means: >
    Reads the entry of the table at the index. If it is present, greater than the
    threshold, and the sample is less than or equal to the threshold, subtracts
    the penalty from the entry, clamping the result to 0. Otherwise, does nothing.

- name: train_penalty
  place: step
  params: {table: table, index: signature, penalty: {int: [1, 127]}}
  means: >
    Reads the entry of the table at the index. If it is absent, sets it to the
    table's maximum. If present, adds the penalty to the entry, clamping the
    result to the table's maximum.
