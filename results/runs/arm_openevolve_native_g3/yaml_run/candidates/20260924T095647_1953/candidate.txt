===== STRUCTURE_DESCRIPTION =====
name: mockingjay_opt
doc: >
  Addresses wrong_bypass by capping RDP at 73 when a line is reused early,
  instantly stopping premature bypasses. A 2-bit confidence counter tracks
  the probability of long reuse; absolute bypass (> 73) requires high confidence
  (>= 2), and relative bypass requires moderate confidence (>= 1).

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {words: {confident_bypass: {rdp: rdp, conf: bypass_conf, index: pc_sig, rdp_thresh: 73, conf_thresh: 2}}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, table: {table: bypass_conf, index: pc_sig, cmp: gte, threshold: 1}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    max: 3
    init: 1

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {counter: {counter: age, cmp: lte, threshold: 73}}
        words: {cap_value: {table: rdp, index: signature, cap: 73}}
      - when: {counter: {counter: age, cmp: gt, threshold: 73}}
        train: {table: bypass_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: lte, threshold: 73}}
        train: {table: bypass_conf, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_conf, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: confident_bypass
  place: condition
  params: {rdp: table, conf: table, index: signature, rdp_thresh: {int: [0, 127]}, conf_thresh: {int: [0, 3]}}
  means: >
    Reads the entry of the rdp table at the index. If it is absent, returns false.
    Otherwise, reads the entry of the conf table at the index (treating an absent
    entry as 0). Returns true if the rdp entry is greater than rdp_thresh and the
    conf entry is greater than or equal to conf_thresh.

- name: cap_value
  place: step
  params: {table: table, index: signature, cap: {int: [0, 127]}}
  means: >
    Reads the entry of the table at the index. If it is present and strictly
    greater than the cap, sets it to the cap.
