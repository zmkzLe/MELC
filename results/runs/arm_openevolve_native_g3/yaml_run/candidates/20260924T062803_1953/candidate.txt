===== STRUCTURE_DESCRIPTION =====
name: mockingjay_conf
doc: >
  Addresses wrong_bypass and should_bypass by adding a bypass confidence table
  and using Exponential Moving Average (EMA) for faster RDP training. A 2-bit
  saturating counter per PC signature tracks the probability of long reuse;
  absolute bypass (> 73) requires high confidence (>= 2), and relative bypass
  requires moderate confidence (>= 1). RDP is trained via EMA (shift 2) to
  quickly adapt to changing reuse distances, and dead lines are penalized by +8
  to reach the bypass threshold faster.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {words: {confident_bypass: {rdp: rdp, conf: bypass_conf, index: pc_sig, rdp_thresh: 73, conf_thresh: 2}}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, table: {table: bypass_conf, index: pc_sig, cmp: gte, threshold: 1}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    max: 3
    init: 1

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {counter: {counter: age, cmp: gt, threshold: 73}}
        train: {table: bypass_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: lte, threshold: 73}}
        train: {table: bypass_conf, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        words: {train_ema: {table: rdp, index: signature, sample: age, scale: 2, shift: 2}}
        free: true
        stop: true
      - words: {train_ema: {table: rdp, index: signature, sample: age, scale: 1, shift: 2}}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        words: {train_penalty: {table: rdp, index: signature, penalty: 8}}
        free: true
    on_replace:
      - train: {table: bypass_conf, index: signature, op: increment}
      - words: {train_penalty: {table: rdp, index: signature, penalty: 8}}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: confident_bypass
  place: condition
  params: {rdp: table, conf: table, index: signature, rdp_thresh: {int: [0, 127]}, conf_thresh: {int: [0, 3]}}
  means: >
    Reads the entry of the rdp table at the index. If it is absent, returns false.
    Otherwise, reads the entry of the conf table at the index (treating an absent
    entry as 0). Returns true if the rdp entry is greater than rdp_thresh and the
    conf entry is greater than or equal to conf_thresh.

- name: train_ema
  place: step
  params: {table: table, index: signature, sample: state, scale: {int: [1, 4]}, shift: {int: [0, 7]}}
  means: >
    Calculates a target value as the sample multiplied by the scale, clamped to
    the table's maximum. Reads the entry of the table at the index. If it is
    absent, sets it to the target value. If present, computes the signed difference
    (target - entry), divides it by 2 to the power of shift (integer division
    rounding towards zero), and adds the result to the entry. The final value is
    clamped to the table's minimum (0) and maximum.

- name: train_penalty
  place: step
  params: {table: table, index: signature, penalty: {int: [1, 127]}}
  means: >
    Reads the entry of the table at the index. If it is absent, sets it to the
    table's maximum. If present, adds the penalty to the entry, clamping the
    result to the table's maximum.
