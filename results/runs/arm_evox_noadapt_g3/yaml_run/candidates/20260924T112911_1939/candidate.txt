===== STRUCTURE_DESCRIPTION =====
name: mockingjay_veto
doc: >
  Mockingjay with Veto: Adds a 4096-entry 1-bit veto table to detect and prevent
  wrong bypasses. If a line is predicted to be bypassed (RDP > 73) but is reused
  in the sampler, its signature is vetoed. Vetoed signatures are not bypassed and
  are inserted with a safe ETR of 0 to prevent premature eviction. The veto is
  cleared if the signature later ages out without reuse. RDP table is 4096 entries,
  and sampler is 256 sets x 8 ways to minimize conflict misses.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {confident_bypass: {rdp_table: rdp, veto_table: veto, index: pc_sig, rdp_thresh: 73}}
    - {confident_exceeds_victim: {rdp_table: rdp, veto_table: veto, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95
    unset: true
  - name: veto
    scope: global
    kind: saturating
    entries: 4096
    width: 1
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 3
    ways: 8
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {table: {table: rdp, index: signature, cmp: gt, threshold: 73}}
        train: {table: veto, index: signature, op: increment}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        train: {table: veto, index: signature, op: decrement}
        free: true
    on_replace:
      - when: {counter: {counter: age, cmp: gt, threshold: 73}}
        train: {table: rdp, index: signature, op: increment}
        train: {table: veto, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {confident_bypass: {rdp_table: rdp, veto_table: veto, index: pc_sig, rdp_thresh: 73}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 0}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access

===== NEW_WORDS =====

- name: confident_bypass
  place: condition
  params: {rdp_table: table, veto_table: table, index: signature, rdp_thresh: {int: [0, 127]}}
  means: >
    True when the entry of rdp_table at the index is greater than rdp_thresh,
    and the entry of veto_table at the index is equal to 0.
    An absent entry reads as 0.

- name: confident_exceeds_victim
  place: condition
  params: {rdp_table: table, veto_table: table, index: signature, div: {int: [1, 32]}}
  means: >
    True when the entry of rdp_table at the index divided by div (integer division)
    is greater than the absolute value of the victim key of the way the victim search chose,
    and the entry of veto_table at the index is equal to 0.
    An absent entry reads as 0. Only valid inside victim.bypass_when.
