name: mockingjay_opt_v6
doc: >
  Mockingjay optimized v6: RDP table at 4096 entries with modulo 4093 to reduce aliasing.
  ETR range expanded to [-15, 15] to utilize the full 5-bit state.
  set_clock and timestamp remain 7 bits (wraps at 128), but max age is safely extended to 126
  since sweep runs on every sampler access, preventing unseen wrap-arounds.
  etr_clock uses 3 bits, wrapping naturally at 7 to trigger aging every 8 accesses.
  Bypass thresholds scaled to the new max RDP of 126 (98 for PREFETCH, 114 for demand).
state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: MAX
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7
    init: ZERO
victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: gt, threshold: 98}}
    - {type_not_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: gt, threshold: 114}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}
signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12
    modulo: 4093
tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 126
    unset: true
samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 3
    ways: 9
    tag_bits: 11
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 125}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 125}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}
procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 7}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 15, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {type_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: gt, threshold: 98}}
      set: {etr: {const: 15}}
      stop: true
    - when: {type_not_in: [PREFETCH], table: {table: rdp, index: pc_sig, cmp: gt, threshold: 114}}
      set: {etr: {const: 15}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}
on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access
on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -15}}
    stop: true
  - run: access
